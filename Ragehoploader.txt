local function findGame(placeId, universeId)

    -- Сначала проверяем Universe ID
    for _, entry in ipairs(CATALOG) do
        if entry.Universe
            and entry.Universe ~= 0
            and entry.Universe == universeId then

            return entry
        end
    end

    -- Потом Place ID
    for _, entry in ipairs(CATALOG) do
        for _, id in ipairs(entry.Places or {}) do
            if id == placeId then
                return entry
            end
        end
    end

    return nil
end

local function loadRemote(url, chunkName)

    local success, source = pcall(function()
        return game:HttpGet(url)
    end)

    if not success then
        warn("[Loader] Failed to download:")
        warn(source)

        return false, source
    end

    if not source or source == "" then
        return false, "Downloaded script is empty"
    end

    local func, compileError = loadstring(
        source,
        "@" .. tostring(chunkName or "RemoteScript")
    )

    if not func then
        warn("[Loader] Compile error:")
        warn(compileError)

        return false, compileError
    end

    local runSuccess, runError = pcall(func)

    if not runSuccess then
        warn("[Loader] Runtime error:")
        warn(runError)

        return false, runError
    end

    return true
end

local function launch(entry)

    print(
        "[Loader] Starting:",
        entry.Name
    )

    local success, err = loadRemote(
        entry.Loader,
        entry.Key
    )

    if not success then
        warn(
            "[Loader] " ..
            entry.Name ..
            " failed: " ..
            tostring(err)
        )
    end

    return success
end

--==================================================
-- DRAG
--==================================================

local UserInputService = game:GetService(
    "UserInputService"
)

local dragging = false
local dragStart
local startPosition

Main.InputBegan:Connect(function(input)

    if input.UserInputType ==
        Enum.UserInputType.MouseButton1 then

        dragging = true

        dragStart = input.Position

        startPosition = Main.Position

    end

end)

UserInputService.InputChanged:Connect(function(input)

    if not dragging then
        return
    end

    if input.UserInputType ==
        Enum.UserInputType.MouseMovement then

        local delta =
            input.Position -
            dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,

            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )

    end

end)

UserInputService.InputEnded:Connect(function(input)

    if input.UserInputType ==
        Enum.UserInputType.MouseButton1 then

        dragging = false

    end

end)
